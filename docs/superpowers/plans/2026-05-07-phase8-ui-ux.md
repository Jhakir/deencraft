> **For agentic workers:** Use superpowers:subagent-driven-development or superpowers:executing-plans. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build all in-game UI panels using Unity's built-in UI Toolkit (uGUI). Wire each panel to the existing backend (FirebaseAuthManager, SessionManager, WorldSaveManager, VitalitySystem, Inventory, AdhanAudioManager, CharacterCustomizer). - Main Menu (logo, Play, Settings, Credits)

```csharp
// Assets/Scripts/UI/SettingsModel.cs
// Pure-C# settings storage. No MonoBehaviour — fully testable in EditMode.
// Reads/writes PlayerPrefs with "dc_" namespace prefix.
using UnityEngine;

namespace DeenCraft.UI
{
    /// <summary>
    /// Holds user-facing settings (volume, graphics, Adhan).
    /// Call <see cref="Load"/> on startup and <see cref="Save"/> after any change.
    /// </summary>
    public sealed class SettingsModel
    {
        // ── PlayerPrefs keys ─────────────────────────────────────────────────
        private const string KeyMusicVol      = "dc_music_vol";
        private const string KeySfxVol        = "dc_sfx_vol";
        private const string KeyGraphicsQual  = "dc_graphics_quality";

        // ── Default values ───────────────────────────────────────────────────
        private const float DefaultMusicVol      = 0.8f;
        private const float DefaultSfxVol        = 1.0f;
        private const int   DefaultGraphicsQual  = 1;   // 0=Low, 1=Medium, 2=High

        // ── Properties (clamped setters) ─────────────────────────────────────
        private float _musicVolume  = DefaultMusicVol;
        private float _sfxVolume    = DefaultSfxVol;

        public float MusicVolume
        {
            get => _musicVolume;
            set => _musicVolume = Mathf.Clamp01(value);
        }

        public float SfxVolume
        {
            get => _sfxVolume;
            set => _sfxVolume = Mathf.Clamp01(value);
        }

        public int GraphicsQuality { get; set; } = DefaultGraphicsQual;

        // ── Lifecycle ────────────────────────────────────────────────────────
        public SettingsModel()
        {
            // Start with defaults — call Load() explicitly if persisted values are wanted.
        }

        public void Load()
        {
            MusicVolume     = PlayerPrefs.GetFloat(KeyMusicVol,     DefaultMusicVol);
            SfxVolume       = PlayerPrefs.GetFloat(KeySfxVol,       DefaultSfxVol);
            GraphicsQuality = PlayerPrefs.GetInt(  KeyGraphicsQual, DefaultGraphicsQual);
        }

        public void Save()
        {
            PlayerPrefs.SetFloat(KeyMusicVol,     _musicVolume);
            PlayerPrefs.SetFloat(KeySfxVol,       _sfxVolume);
            PlayerPrefs.SetInt(  KeyGraphicsQual, GraphicsQuality);
            PlayerPrefs.Save();
        }
    }
}
```

```csharp
// Assets/Scripts/UI/HudViewModel.cs
// Pure-C# helpers that convert VitalityData values into HUD display values.
// No MonoBehaviour — fully testable in EditMode.
using UnityEngine;
using DeenCraft;

namespace DeenCraft.UI
{
    /// <summary>
    /// Converts raw game data into HUD-ready display values.
    /// All methods are pure functions — no state.
    /// </summary>
    public static class HudViewModel
    {
        // ── Health ────────────────────────────────────────────────────────────

        /// <summary>
        /// Converts health (0–MaxHealth) to a heart count (0–10).
        /// Full hearts only — partial hearts are rounded down.
        /// </summary>
        public static int HealthToHearts(float health)
        {
            float clamped = Mathf.Clamp(health, 0f, GameConstants.MaxHealth);
            return Mathf.FloorToInt(clamped / 2f);
        }

        // ── Hunger ────────────────────────────────────────────────────────────

        /// <summary>
        /// Returns hunger as a 0–1 fraction for use in a fill-bar Image.
        /// </summary>
        public static float HungerFraction(float hunger)
            => Mathf.Clamp01(hunger / GameConstants.MaxHunger);

        // ── Day/Time ─────────────────────────────────────────────────────────

        /// <summary>
        /// Converts normalizedTime (0=midnight, 0.25=dawn, 0.5=noon, 0.75=dusk)
        /// to a short human-readable label.
        /// </summary>
        public static string TimeLabel(float normalizedTime)
        {
            float t = Mathf.Clamp01(normalizedTime);
            if (t < 0.125f || t >= 0.875f) return "Night";
            if (t < 0.375f) return "Morning";
            if (t < 0.625f) return "Midday";
            return "Evening";
        }
    }
}
```

```csharp
// Assets/Scripts/UI/HudController.cs
// Polls VitalitySystem and Inventory each frame and updates HUD UI elements.
// Attach to the HUD Canvas root GameObject.
// Wire all [SerializeField] slots in the Inspector.
using UnityEngine;
using UnityEngine.UI;
using DeenCraft.Player;
using DeenCraft;

namespace DeenCraft.UI
{
    /// <summary>
    /// Updates the in-game HUD every frame:
    /// health hearts, hunger bar, hotbar slots, day/time label.
    ///
    /// Attach to the HUD root CanvasGroup. Wire Inspector references.
    /// </summary>
    public sealed class HudController : MonoBehaviour
    {
        // ── Data sources ─────────────────────────────────────────────────────
        [SerializeField] private VitalitySystem _vitalitySystem;
        [SerializeField] private InventoryHolder _inventoryHolder;

        // ── Heart display ─────────────────────────────────────────────────────
        [SerializeField] private Image[] _heartIcons;      // 10 icons
        [SerializeField] private Sprite  _heartFull;
        [SerializeField] private Sprite  _heartEmpty;

        // ── Hunger bar ────────────────────────────────────────────────────────
        [SerializeField] private Image _hungerFill;        // Image.fillAmount

        // ── Hotbar ────────────────────────────────────────────────────────────
        [SerializeField] private HotbarSlotUI[] _hotbarSlots;   // 9 slots
        [SerializeField] private Image          _hotbarSelector; // moves to active slot

        // ── Day/time label ────────────────────────────────────────────────────
        [SerializeField] private Text _timeLabel;

        private void Update()
        {
            if (_vitalitySystem != null)
            {
                UpdateHearts(_vitalitySystem.Health);
                UpdateHunger(_vitalitySystem.Hunger);
            }

            if (_inventoryHolder != null)
                UpdateHotbar(_inventoryHolder.Inventory);

            UpdateTimeLabel();
        }

        private void UpdateHearts(float health)
        {
            int filledHearts = HudViewModel.HealthToHearts(health);
            for (int i = 0; i < _heartIcons.Length; i++)
            {
                if (_heartIcons[i] == null) continue;
                _heartIcons[i].sprite = i < filledHearts ? _heartFull : _heartEmpty;
            }
        }

        private void UpdateHunger(float hunger)
        {
            if (_hungerFill == null) return;
            _hungerFill.fillAmount = HudViewModel.HungerFraction(hunger);
        }

        private void UpdateHotbar(Inventory inventory)
        {
            for (int i = 0; i < _hotbarSlots.Length; i++)
            {
                if (_hotbarSlots[i] == null) continue;
                _hotbarSlots[i].Refresh(inventory.GetHotbarSlot(i));
            }

            if (_hotbarSelector != null)
            {
                float slotWidth = ((RectTransform)_hotbarSelector.transform.parent).rect.width
                                  / GameConstants.HotbarSlots;
                var rt = (RectTransform)_hotbarSelector.transform;
                rt.anchoredPosition = new Vector2(inventory.SelectedSlot * slotWidth, 0f);
            }
        }

        private void UpdateTimeLabel()
        {
            if (_timeLabel == null || DeenCraft.World.DayNightCycle.Instance == null) return;
            _timeLabel.text = HudViewModel.TimeLabel(
                DeenCraft.World.DayNightCycle.Instance.NormalizedTime);
        }
    }
}
```

```csharp
// Assets/Scripts/UI/ChildProfilePickerPanel.cs
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;
using DeenCraft.Auth;
using DeenCraft.Auth.Models;

namespace DeenCraft.UI
{
    /// <summary>
    /// Shows all child profiles under the active parent account.
    /// Child taps their profile → PIN entry (if required) → world select.
    /// </summary>
    public sealed class ChildProfilePickerPanel : MonoBehaviour
    {
        [SerializeField] private Transform      _profileContainer;   // scroll view content
        [SerializeField] private GameObject     _profileButtonPrefab;
        [SerializeField] private GameObject     _pinEntryPanel;
        [SerializeField] private InputField     _pinInputField;
        [SerializeField] private Text           _errorLabel;
        [SerializeField] private GameObject     _worldSelectPanel;

        private List<ChildProfile> _profiles;
        private ChildProfile       _selectedProfile;

        private async void OnEnable()
        {
            _errorLabel.text = "";
            _profiles = await FirebaseAuthManager.Instance.LoadChildProfilesAsync();
            BuildProfileButtons();
        }

        private void BuildProfileButtons()
        {
            foreach (Transform child in _profileContainer)
                Destroy(child.gameObject);

            foreach (var profile in _profiles)
            {
                var go  = Instantiate(_profileButtonPrefab, _profileContainer);
                var btn = go.GetComponent<Button>();
                var lbl = go.GetComponentInChildren<Text>();
                if (lbl != null) lbl.text = profile.username;

                var captured = profile;
                btn.onClick.AddListener(() => OnProfileSelected(captured));
            }
        }

        private void OnProfileSelected(ChildProfile profile)
        {
            _selectedProfile = profile;

            if (!string.IsNullOrEmpty(profile.pinHash))
            {
                _pinEntryPanel.SetActive(true);
                _pinInputField.text = "";
            }
            else
            {
                ActivateProfile("");
            }
        }

        public async void OnPinConfirmed()
        {
            if (_selectedProfile == null) return;
            _errorLabel.text = "";

            try
            {
                await FirebaseAuthManager.Instance.ActivateChildAsync(
                    _selectedProfile, _pinInputField.text);
                _pinEntryPanel.SetActive(false);
                ShowWorldSelect();
            }
            catch (System.UnauthorizedAccessException)
            {
                _errorLabel.text = "Incorrect PIN — try again.";
            }
            catch (System.Exception ex)
            {
                _errorLabel.text = $"Error: {ex.Message}";
            }
        }

        private async void ActivateProfile(string pin)
        {
            try
            {
                await FirebaseAuthManager.Instance.ActivateChildAsync(_selectedProfile, pin);
                ShowWorldSelect();
            }
            catch (System.Exception ex)
            {
                _errorLabel.text = $"Error: {ex.Message}";
            }
        }

        private void ShowWorldSelect()
        {
            gameObject.SetActive(false);
            if (_worldSelectPanel != null) _worldSelectPanel.SetActive(true);
        }
    }
}
```

```csharp
// Assets/Scripts/UI/WorldSelectPanel.cs
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;
using DeenCraft.Auth;
using DeenCraft.Auth.Models;
using DeenCraft.World;

namespace DeenCraft.UI
{
    /// <summary>
    /// Shows the active child's saved worlds.
    /// "New World" → calls WorldSaveManager.NewWorldAsync.
    /// Tap existing → calls WorldSaveManager.LoadWorldAsync.
    /// </summary>
    public sealed class WorldSelectPanel : MonoBehaviour
    {
        [SerializeField] private Transform    _worldContainer;
        [SerializeField] private GameObject   _worldButtonPrefab;
        [SerializeField] private InputField   _newWorldNameField;
        [SerializeField] private Text         _errorLabel;
        [SerializeField] private WorldSaveManager _worldSaveManager;

        private List<WorldSaveData> _saves;

        private async void OnEnable()
        {
            _errorLabel.text = "";
            // Backend returns saves for the active child via FirebaseAuthManager internally
            _saves = new List<WorldSaveData>();
            // Note: ListWorldSavesAsync requires parent uid + child id — delegate to AuthManager
            try
            {
                // Use the IDataBackend via FirebaseAuthManager path
                _saves = await FirebaseAuthManager.Instance.ListWorldSavesAsync();
            }
            catch (System.Exception ex)
            {
                _errorLabel.text = $"Failed to load worlds: {ex.Message}";
            }
            BuildWorldButtons();
        }

        private void BuildWorldButtons()
        {
            foreach (Transform child in _worldContainer)
                Destroy(child.gameObject);

            foreach (var save in _saves)
            {
                var go  = Instantiate(_worldButtonPrefab, _worldContainer);
                var btn = go.GetComponent<Button>();
                var lbl = go.GetComponentInChildren<Text>();
                if (lbl != null)
                    lbl.text = $"{save.worldName}\nDay {save.dayCount}";

                var id = save.id;
                btn.onClick.AddListener(async () =>
                {
                    gameObject.SetActive(false);
                    await _worldSaveManager.LoadWorldAsync(id);
                });
            }
        }

        public async void OnNewWorldPressed()
        {
            string name = _newWorldNameField != null
                ? _newWorldNameField.text.Trim()
                : "My World";
            if (string.IsNullOrEmpty(name)) name = "My World";

            gameObject.SetActive(false);
            await _worldSaveManager.NewWorldAsync(name);
        }
    }
}
```

```csharp
                _skinToneButtons[i]?.onClick.AddListener(() => OnSkinTone(idx));
            }
            for (int i = 0; i < _headwearButtons.Length; i++)
            {
                var idx = i;
                _headwearButtons[i]?.onClick.AddListener(() => OnHeadwear(idx));
            }
            for (int i = 0; i < _clothingStyleButtons.Length; i++)
            {
                var idx = i;
                _clothingStyleButtons[i]?.onClick.AddListener(() => OnClothingStyle(idx));
            }
            for (int i = 0; i < _colorSwatches.Length; i++)
            {
                var idx = i;
                _colorSwatches[i]?.onClick.AddListener(() => OnColor(idx));
            }
        }

        private void OnSkinTone(int index)
        {
            var a = _characterCustomizer.Appearance;
            a.SkinTone = (SkinTone)index;
            ApplyAndSave(a);
        }

        private void OnHeadwear(int index)
        {
            var a = _characterCustomizer.Appearance;
            a.HeadwearType = (HeadwearType)index;
            ApplyAndSave(a);
        }

        private void OnClothingStyle(int index)
        {
            var a = _characterCustomizer.Appearance;
            a.ClothingStyle = (ClothingStyle)index;
            ApplyAndSave(a);
        }

        private void OnColor(int index)
        {
            if (index < 0 || index >= _presetColors.Length) return;
            var a = _characterCustomizer.Appearance;
            a.ClothingColor = _presetColors[index];
            ApplyAndSave(a);
        }

        private void ApplyAndSave(CharacterAppearance appearance)
        {
            _characterCustomizer.SetAppearance(appearance);
            _characterCustomizer.SaveToPrefs();

            // Persist to child profile so WorldSaveManager picks it up on next save
            if (SessionManager.IsChildActive)
                SessionManager.ActiveChild.character = CharacterDataBridge.ToData(appearance);
        }
    }
}
```

```csharp
namespace DeenCraft.UI

    /// <summary>

            _pausePanel.SetActive(paused);

        }

        {
            SetPaused(false);
        }

        {
            if (_worldSaveManager != null)
                await _worldSaveManager.SaveAsync();
        }

        public async void OnSaveAndQuitPressed()
        {
            if (_worldSaveManager != null)
                await _worldSaveManager.SaveAndQuitAsync();
            Time.timeScale = 1f;
            SceneManager.LoadScene("MainMenu");
        }

        public void OnSettingsPressed()
        {
            _pausePanel.SetActive(false);
            _settingsPanel.SetActive(true);
        }
    }
```

### SettingsPanel

```csharp
// Assets/Scripts/UI/SettingsPanel.cs
// Reads/writes SettingsModel and delegates specific settings to AdhanAudioManager.
using UnityEngine;
using UnityEngine.Audio;
using UnityEngine.UI;
using DeenCraft.World;

namespace DeenCraft.UI
{
    /// <summary>
    /// Settings panel controller. Attach to the Settings Canvas root.
    /// Wire Inspector references. Call from MainMenuController or PauseMenuController.
    /// </summary>
    public sealed class SettingsPanel : MonoBehaviour
    {
        // ── Inspector ────────────────────────────────────────────────────────
        [SerializeField] private AudioMixer  _audioMixer;           // "MasterMix"
        [SerializeField] private Slider      _musicSlider;          // 0–1
        [SerializeField] private Slider      _sfxSlider;            // 0–1
        [SerializeField] private Toggle      _adhanToggle;
        [SerializeField] private Dropdown    _adhanClipDropdown;
        [SerializeField] private Dropdown    _graphicsDropdown;     // Low / Medium / High
        [SerializeField] private Button      _backButton;

        [SerializeField] private AdhanAudioManager _adhanAudioManager;

        // ── Back destination ─────────────────────────────────────────────────
        [SerializeField] private GameObject  _returnPanel;          // panel to re-show on Back

        private SettingsModel _model;

        private void OnEnable()
        {
            _model = new SettingsModel();
            _model.Load();
            RefreshUI();
            WireListeners();
        }

        private void OnDisable()
        {
            _model.Save();
        }

        private void RefreshUI()
        {
            if (_musicSlider       != null) _musicSlider.value       = _model.MusicVolume;
            if (_sfxSlider         != null) _sfxSlider.value         = _model.SfxVolume;
            if (_adhanToggle       != null && _adhanAudioManager != null)
                _adhanToggle.isOn = _adhanAudioManager.IsEnabled;
            if (_graphicsDropdown  != null) _graphicsDropdown.value  = _model.GraphicsQuality;
        }

        private void WireListeners()
        {
            _musicSlider?.onValueChanged.AddListener(OnMusicVolume);
            _sfxSlider?.onValueChanged.AddListener(OnSfxVolume);
            _adhanToggle?.onValueChanged.AddListener(OnAdhanToggle);
            _adhanClipDropdown?.onValueChanged.AddListener(OnAdhanClip);
            _graphicsDropdown?.onValueChanged.AddListener(OnGraphicsQuality);
            _backButton?.onClick.AddListener(OnBack);
        }

        private void OnMusicVolume(float value)
        {
            _model.MusicVolume = value;
            if (_audioMixer != null)
                _audioMixer.SetFloat("MusicVolume", Mathf.Log10(Mathf.Max(value, 0.0001f)) * 20f);
        }

        private void OnSfxVolume(float value)
        {
            _model.SfxVolume = value;
            if (_audioMixer != null)
                _audioMixer.SetFloat("SFXVolume", Mathf.Log10(Mathf.Max(value, 0.0001f)) * 20f);
        }

        private void OnAdhanToggle(bool isOn)
        {
            if (_adhanAudioManager != null)
                _adhanAudioManager.SetEnabled(isOn);
        }

        private void OnAdhanClip(int index)
        {
            if (_adhanAudioManager != null)
                _adhanAudioManager.SetClipIndex(index);
        }

        private void OnGraphicsQuality(int index)
        {
            _model.GraphicsQuality = index;
            QualitySettings.SetQualityLevel(index, applyExpensiveChanges: true);
        }

        private void OnBack()
        {
            _model.Save();
            gameObject.SetActive(false);
            if (_returnPanel != null) _returnPanel.SetActive(true);
        }
    }
}
```

## Conventions Reminder
- `[SerializeField]` not `public` for Inspector fields
- `async void` only for Unity event handlers (button callbacks) — never `async void` for anything called from tests
- uGUI: `Text` from `UnityEngine.UI`, `Image`, `Button`, `Slider`, `Toggle`, `Dropdown`, `InputField`
- Volume → AudioMixer: use `Mathf.Log10(vol) * 20f` (decibel conversion) — never set linear values directly on an AudioMixer
- `Time.timeScale = 0f` to pause, `= 1f` to resume
